      \xeCJKsetup{AutoFakeBold=2.5}}}}

% 顶部：左边姓名 / 求职意向 / 联系方式，右上角可选证件照；随后是第一个分节
% #1 姓名, #2 求职意向（可空）, #3 联系方式, #4 证件照路径（可空）, #5 首个分节名
\newcommand{\resumeheader}[5]{%
  \noindent
  \ifstrempty{#4}{%
    \begin{minipage}[t]{\linewidth}%
  }{%
    \begin{minipage}[t]{\linewidth-76bp}%
  }%
    \vspace{0pt}\raggedright
    {\bfseries\fontsize{19bp}{23bp}\selectfont #1}\par
    \ifstrempty{#2}{}{\vspace{3bp}{\color{black!62}#2}\par}%
    \vspace{4bp}{\color{black!85}#3}\par
  \end{minipage}%
  \ifstrempty{#4}{}{\hfill\begin{minipage}[t]{64bp}\vspace{0pt}\includegraphics[width=64bp,height=80bp,keepaspectratio=false]{#4}\end{minipage}}%
  \par\vspace{2bp}%
  \resumesection{#5}}

% 分节：加粗标题 + 通栏细线
\newif\ifafterpoint
\afterpointfalse
\newcommand{\resumesection}[1]{%
  \par\Needspace{48bp}%
  \vspace{8bp}%
  {\bfseries\fontsize{11.5bp}{14bp}\selectfont #1}\par
  \vspace{1.6bp}\hrule height 0.7bp\relax\vspace{3.2bp}%
  \global\afterpointfalse}

% 条目头：左侧「机构 - 角色」加粗，右侧时间
\newcommand{\entry}[2]{%
  \par\Needspace{36bp}%
  \ifafterpoint\vspace{2.4bp}\fi
  \noindent\textbf{#1}\hfill{\color{black!85}#2}\par
  \global\afterpointfalse}

% 条目下的一句概述（不带圆点）
\newcommand{\summaryline}[1]{\par\noindent #1\par}

% 不带圆点的一行，可选加粗前缀（教育背景的 GPA、主修课程等）
\newcommand{\plainline}[2]{\par\noindent\ifstrempty{#1}{}{\textbf{#1}}#2\par}

% 要点：圆点 + 可选加粗小标题 + 正文
\newcommand{\point}[2]{%
  \par\Needspace{14bp}%
  \begingroup
    \leftskip=11bp\relax
    \noindent\llap{\raisebox{.12ex}{\scalebox{.75}{$\bullet$}}\hspace{5bp}}%
    \ifstrempty{#1}{}{\textbf{#1}}#2\par
  \endgroup
  \global\afterpointtrue}
