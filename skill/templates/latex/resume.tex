% 经历库 · 简历模板（单栏 A4，参考 sqs-resume 风格：紧凑留白 + 灰底黑条分节栏）
% 由 scripts/fill_tex.py 填充 %%占位符%%
% 编译：xelatex（需要 ctex 宏包）
\documentclass[UTF8]{ctexart}
\usepackage[a4paper,left=22.5bp,right=22.17bp,top=19.5bp,bottom=22.5bp]{geometry}
\usepackage{xcolor,tikz,needspace,etoolbox,graphicx}
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{1.4bp}
\setlength{\topskip}{0pt}
\setlength{\emergencystretch}{0pt}
\clubpenalty=10000
\widowpenalty=10000

\newcommand{\resumebodyfont}{\fontsize{9bp}{10.5bp}\selectfont}
\AtBeginDocument{\resumebodyfont\raggedright}

% 顶部个人信息 + 首个分节栏（可选右上角证件照）
% #1 姓名, #2 副标题/求职意向（可空）, #3 联系方式, #4 证件照路径（可空）, #5 首个分节名
\newcommand{\resumeheader}[5]{%
  \ifstrempty{#4}{%
    \noindent\textbf{\fontsize{15bp}{18bp}\selectfont #1}%
    \ifstrempty{#2}{}{\quad{\color{black!60}#2}}\par
    \vspace{2bp}
    \noindent{\color{black!75}#3}\par
    \vspace{4bp}
    \resumebar{#5}%
  }{%
    \noindent\begin{tikzpicture}[x=1bp,y=1bp,baseline=0bp]
      \path[use as bounding box] (0,0) rectangle (551.25,76);
      \node[anchor=north east,inner sep=0pt] at (551.25,76) {\includegraphics[width=60bp,height=72bp]{#4}};
      \node[anchor=base west,inner sep=0pt,font=\bfseries\fontsize{15bp}{18bp}\selectfont] at (0,44) {#1};
      \ifstrempty{#2}{}{\node[anchor=base west,inner sep=0pt,font=\resumebodyfont,text=black!60] at (0,59) {#2};}
      \node[anchor=base west,inner sep=0pt,font=\resumebodyfont,text=black!80] at (0,24) {#3};
      \fill[black!5] (0,0) rectangle (479.25,18);
      \fill[black] (0,0) rectangle (3,18);
      \node[anchor=base west,inner sep=0pt,font=\bfseries\fontsize{10.5bp}{12bp}\selectfont] at (15,5.25) {#5};
    \end{tikzpicture}\par\vspace{1.5bp}%
  }%
}

% 18bp 灰底 + 3bp 左侧黑条分节栏
\newcommand{\resumebar}[1]{%
  \begingroup
  \setlength{\fboxsep}{0pt}%
  \noindent\colorbox{black!5}{%
    \makebox[\linewidth][l]{%
      \rule{3bp}{18bp}\hspace{12bp}%
      \raisebox{5.25bp}{\bfseries\fontsize{10.5bp}{12bp}\selectfont #1}%
      \hfill}}%
  \par\endgroup\vspace{1.5bp}}

\newif\ifafterpoint
\afterpointfalse

\newcommand{\resumesection}[1]{%
  \par\Needspace{55bp}%
  \ifdim\pagetotal=0pt\relax\vspace*{10bp}\fi
  \vspace{5.7bp}\nointerlineskip
  \resumebar{#1}%
  \global\afterpointfalse}

% 条目头：左侧加粗，右侧时间
\newcommand{\entry}[2]{%
  \par\Needspace{45bp}%
  \ifafterpoint\ifstrempty{#2}{\vspace{3bp}}{\vspace{6bp}}\fi
  \noindent\textbf{#1}\hfill{\color{black!70}#2}\par
  \global\afterpointfalse}

% 要点：可选加粗前缀 + 正文
\newcommand{\point}[2]{%
  \par\Needspace{21bp}%
  \begingroup
    \leftskip=15bp\relax
    \noindent\llap{\textbullet\hspace{7bp}}%
    \ifstrempty{#1}{}{\textbf{#1}}#2\par
  \endgroup
  \global\afterpointtrue}
